Docker local shell = same two modes as Electron.

  CHAPTERDESK_STORAGE=sqlite-all     # default: everything in mounted SQLite
  CHAPTERDESK_STORAGE=chats-idb      # Electron-with-chats-in-IDB
  CHAPTERDESK_STORAGE=chats-params-idb
  CHAPTERDESK_REQUIRE_AUTH=1         # optional; default is skip login like Electron

package.json:

  "docker:dev": "mkdir -p chat-server/data && docker build -f Dockerfile.dev -t chapterdesk-dev . && docker run --env-file .env --rm --name chapterdesk-dev -p 8080:8080 -p 9229:9229 -v ./chat-server/data:/app/data -e CHAPTERDESK_DOCKER=1 -e CHAPTERDESK_RUNTIME=docker-dev -e CHAPTERDESK_STORAGE=sqlite-all chapterdesk-dev"

  "docker:dev:idb": "mkdir -p chat-server/data && docker build -f Dockerfile.dev -t chapterdesk-dev . && docker run --env-file .env --rm --name chapterdesk-dev -p 8080:8080 -p 9229:9229 -v ./chat-server/data:/app/data -e CHAPTERDESK_DOCKER=1 -e CHAPTERDESK_RUNTIME=docker-dev -e CHAPTERDESK_STORAGE=chats-idb chapterdesk-dev"

Electron chats in IDB:

  CHAPTERDESK_STORAGE=chats-idb npm run electron
  (electron/main.js already forwards CHAPTERDESK_STORAGE)
